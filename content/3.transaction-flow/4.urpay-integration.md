# URPAY Integration

URPAY is a popular digital wallet solution in Saudi Arabia that provides seamless mobile payment experiences. This guide covers the integration of URPAY and other digital wallet solutions with Al Rajhi Bank Payment Gateway.

## Overview

URPAY integration allows customers to pay using their registered mobile numbers, providing a quick and secure payment method without entering card details. This is particularly popular for mobile-first businesses and recurring payment scenarios.

### Key Benefits

- **One-Click Payments**: Customers pay using just their mobile number
- **Enhanced Security**: No card details required from customers
- **Mobile Optimized**: Perfect for mobile applications and responsive websites
- **Quick Checkout**: Reduced checkout time and cart abandonment
- **Local Preference**: Popular payment method in Saudi Arabia

### URPAY Wallet Payment

```php
<?php
/**
 * URPAY Wallet Integration
 */
class URPAYIntegration {
    private $gateway;
    
    public function __construct($gateway) {
        $this->gateway = $gateway;
    }
    
    /**
     * Initiate URPAY payment
     */
    public function initiateURPAYPayment($paymentData) {
        // Validate URPAY mobile number (9 digits)
        if (!$this->validateURPAYMobile($paymentData['mobile_number'])) {
            return [
                'success' => false,
                'error' => 'Invalid URPAY mobile number format',
                'error_code' => 'INVALID_MOBILE'
            ];
        }
        
        $urpayParams = [
            'amt' => $paymentData['amount'],
            'action' => '1', // Purchase
            'password' => $this->gateway->getCredentials()['password'],
            'id' => $this->gateway->getCredentials()['tranportal_id'],
            'currencyCode' => '682', // SAR
            'trackId' => $paymentData['order_id'],
            'mobileNumber' => $paymentData['mobile_number'], // 9-digit URPAY mobile
            'responseURL' => $paymentData['callback_url'],
            'errorURL' => $paymentData['error_url'] ?? $paymentData['callback_url'],
            
            // Optional parameters
            'udf1' => $paymentData['customer_id'] ?? '',
            'udf2' => $paymentData['customer_email'] ?? '',
            'udf3' => $paymentData['order_reference'] ?? '',
            'udf4' => $paymentData['product_category'] ?? '',
            'udf5' => $paymentData['promotion_code'] ?? ''
        ];
        
        try {
            $response = $this->gateway->createURPAYPayment($urpayParams);
            
            if ($response['success']) {
                // Store payment session for verification
                $this->storePaymentSession($response['paymentId'], $paymentData);
                
                return [
                    'success' => true,
                    'payment_id' => $response['paymentId'],
                    'payment_url' => $response['paymentURL'],
                    'mobile_number' => $paymentData['mobile_number'],
                    'amount' => $paymentData['amount']
                ];
            } else {
                return [
                    'success' => false,
                    'error' => $response['error'],
                    'error_code' => $response['error_code']
                ];
            }
        } catch (Exception $e) {
            return [
                'success' => false,
                'error' => 'URPAY payment initialization failed: ' . $e->getMessage(),
                'error_code' => 'INITIALIZATION_ERROR'
            ];
        }
    }
    
    /**
     * Handle URPAY payment callback
     */
    public function handleURPAYCallback($callbackData) {
        try {
            $result = $this->gateway->handleCallback($callbackData);
            
            if ($result['success']) {
                $transactionData = $result['data'];
                
                // Verify URPAY payment
                $verification = $this->verifyURPAYPayment($result['paymentId']);
                
                if ($verification['success']) {
                    // Process successful payment
                    $this->processSuccessfulURPAYPayment($transactionData);
                    
                    return [
                        'success' => true,
                        'transaction_id' => $transactionData['tranid'],
                        'amount' => $transactionData['amt'],
                        'mobile_number' => $transactionData['mobileNumber'] ?? '',
                        'status' => 'completed'
                    ];
                }
            }
            
            // Handle failed payment
            $this->processFailedURPAYPayment($result);
            
            return [
                'success' => false,
                'error' => $result['error'] ?? 'URPAY payment failed',
                'error_code' => $result['error_code'] ?? 'PAYMENT_FAILED'
            ];
            
        } catch (Exception $e) {
            return [
                'success' => false,
                'error' => 'URPAY callback processing failed: ' . $e->getMessage(),
                'error_code' => 'CALLBACK_ERROR'
            ];
        }
    }
    
    /**
     * Validate URPAY mobile number format
     */
    private function validateURPAYMobile($mobileNumber) {
        // URPAY mobile numbers are 9 digits (Saudi mobile without country code)
        return preg_match('/^[0-9]{9}$/', $mobileNumber);
    }
    
    private function storePaymentSession($paymentId, $paymentData) {
        $_SESSION['urpay_payment_' . $paymentId] = [
            'order_id' => $paymentData['order_id'],
            'amount' => $paymentData['amount'],
            'mobile_number' => $paymentData['mobile_number'],
            'customer_id' => $paymentData['customer_id'] ?? '',
            'timestamp' => time()
        ];
    }
    
    private function verifyURPAYPayment($paymentId) {
        return $this->gateway->getTransactionStatus($paymentId);
    }
    
    private function processSuccessfulURPAYPayment($transactionData) {
        // Update order status, send notifications, etc.
        // Implementation depends on your business logic
    }
    
    private function processFailedURPAYPayment($result) {
        // Handle failed payment logic
        // Implementation depends on your business logic
    }
}
```

```php
/**
 * Apple Pay Integration with ARB Payment Gateway
 */
class ApplePayIntegration {
    private $gateway;
    private $merchantId;
    private $merchantDomain;
    
    public function __construct($gateway, $merchantId, $merchantDomain) {
        $this->gateway = $gateway;
        $this->merchantId = $merchantId;
        $this->merchantDomain = $merchantDomain;
    }
    
    /**
     * Initialize Apple Pay session
     */
    public function initializeApplePaySession($paymentData) {
        $applePayParams = [
            'amt' => $paymentData['amount'],
            'action' => '1', // Purchase
            'password' => $this->gateway->getCredentials()['password'],
            'id' => $this->gateway->getCredentials()['tranportal_id'],
            'currencyCode' => '682', // SAR
            'trackId' => $paymentData['order_id'],
            'responseURL' => $paymentData['callback_url'],
            'errorURL' => $paymentData['error_url'] ?? $paymentData['callback_url'],
            
            // Apple Pay specific parameters
            'apple_pay_merchant_id' => $this->merchantId,
            'apple_pay_domain' => $this->merchantDomain,
            'payment_method' => 'APPLE_PAY'
        ];
        
        return $this->gateway->createApplePaySession($applePayParams);
    }
    
    /**
     * Process Apple Pay payment
     */
    public function processApplePayPayment($applePayToken, $paymentData) {
        $paymentParams = [
            'amt' => $paymentData['amount'],
            'action' => '1',
            'password' => $this->gateway->getCredentials()['password'],
            'id' => $this->gateway->getCredentials()['tranportal_id'],
            'currencyCode' => '682',
            'trackId' => $paymentData['order_id'],
            'apple_pay_token' => $applePayToken,
            'responseURL' => $paymentData['callback_url']
        ];
        
        return $this->gateway->processApplePayPayment($paymentParams);
    }
}
```

### Apple Pay JavaScript Implementation

```javascript
// Apple Pay JavaScript integration
class ApplePayHandler {
    constructor(merchantId, merchantDomain) {
        this.merchantId = merchantId;
        this.merchantDomain = merchantDomain;
    }
    
    // Check if Apple Pay is available
    isApplePayAvailable() {
        return window.ApplePaySession && 
               ApplePaySession.canMakePayments() &&
               ApplePaySession.supportsVersion(3);
    }
    
    // Initialize Apple Pay payment
    async initiateApplePayPayment(paymentData) {
        if (!this.isApplePayAvailable()) {
            throw new Error('Apple Pay is not available');
        }
        
        const paymentRequest = {
            countryCode: 'SA',
            currencyCode: 'SAR',
            supportedNetworks: ['visa', 'masterCard', 'amex'],
            merchantCapabilities: ['supports3DS'],
            total: {
                label: paymentData.merchantName || 'Al Rajhi Bank Merchant',
                amount: paymentData.amount
            },
            lineItems: paymentData.lineItems || []
        };
        
        const session = new ApplePaySession(3, paymentRequest);
        
        session.onvalidatemerchant = async (event) => {
            try {
                const merchantSession = await this.validateMerchant(event.validationURL);
                session.completeMerchantValidation(merchantSession);
            } catch (error) {
                console.error('Merchant validation failed:', error);
                session.abort();
            }
        };
        
        session.onpaymentauthorized = async (event) => {
            try {
                const result = await this.processPayment(event.payment, paymentData);
                
                if (result.success) {
                    session.completePayment(ApplePaySession.STATUS_SUCCESS);
                    this.handlePaymentSuccess(result);
                } else {
                    session.completePayment(ApplePaySession.STATUS_FAILURE);
                    this.handlePaymentError(result.error);
                }
            } catch (error) {
                console.error('Payment processing failed:', error);
                session.completePayment(ApplePaySession.STATUS_FAILURE);
                this.handlePaymentError(error.message);
            }
        };
        
        session.begin();
    }
    
    // Validate merchant with Apple
    async validateMerchant(validationURL) {
        const response = await fetch('/apple-pay/validate-merchant', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify({
                validationURL: validationURL,
                merchantId: this.merchantId,
                domain: this.merchantDomain
            })
        });
        
        if (!response.ok) {
            throw new Error('Merchant validation failed');
        }
        
        return await response.json();
    }
    
    // Process Apple Pay payment
    async processPayment(payment, paymentData) {
        const response = await fetch('/apple-pay/process-payment', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json',
            },
            body: JSON.stringify({
                applePayToken: payment.token,
                paymentData: paymentData
            })
        });
        
        return await response.json();
    }
    
    // Handle successful payment
    handlePaymentSuccess(result) {
        console.log('Apple Pay payment successful:', result);
        // Redirect to success page or show success message
        window.location.href = '/payment-success?transaction_id=' + result.transactionId;
    }
    
    // Handle payment error
    handlePaymentError(error) {
        console.error('Apple Pay payment failed:', error);
        // Show error message to user
        alert('Payment failed: ' + error);
    }
}

// Usage
document.addEventListener('DOMContentLoaded', function() {
    const applePayHandler = new ApplePayHandler('merchant.com.yourcompany.app', 'yoursite.com');
    
    const applePayButton = document.getElementById('apple-pay-button');
    
    if (applePayHandler.isApplePayAvailable()) {
        applePayButton.style.display = 'block';
        
        applePayButton.addEventListener('click', function() {
            const paymentData = {
                amount: '100.00',
                orderId: 'ORDER_123',
                merchantName: 'Your Store Name',
                lineItems: [
                    {
                        label: 'Product Name',
                        amount: '100.00'
                    }
                ]
            };
            
            applePayHandler.initiateApplePayPayment(paymentData);
        });
    } else {
        applePayButton.style.display = 'none';
    }
});
```

## Digital Wallet Best Practices

### Security Considerations

```php
/**
 * Digital wallet security utilities
 */
class DigitalWalletSecurity {
    
    /**
     * Validate mobile number for various wallet providers
     */
    public static function validateWalletMobile($mobileNumber, $walletType) {
        switch (strtoupper($walletType)) {
            case 'URPAY':
                // URPAY: 9-digit Saudi mobile number
                return preg_match('/^[0-9]{9}$/', $mobileNumber) && 
                       substr($mobileNumber, 0, 1) === '5';
                
            case 'STCPAY':
                // STC Pay: Similar format to URPAY
                return preg_match('/^[0-9]{9}$/', $mobileNumber) && 
                       substr($mobileNumber, 0, 1) === '5';
                
            default:
                return false;
        }
    }
    
    /**
     * Generate secure wallet transaction reference
     */
    public static function generateWalletTransactionRef($walletType) {
        $prefix = strtoupper($walletType) . '_';
        $timestamp = time();
        $random = bin2hex(random_bytes(8));
        
        return $prefix . $timestamp . '_' . $random;
    }
    
    /**
     * Log wallet transaction attempts
     */
    public static function logWalletTransaction($walletType, $mobileNumber, $amount, $status) {
        $logEntry = [
            'timestamp' => date('Y-m-d H:i:s'),
            'wallet_type' => $walletType,
            'mobile_number' => self::maskMobileNumber($mobileNumber),
            'amount' => $amount,
            'status' => $status,
            'ip_address' => $_SERVER['REMOTE_ADDR'] ?? 'Unknown'
        ];
        
        error_log('Wallet Transaction: ' . json_encode($logEntry), 3, 'wallet_transactions.log');
    }
    
    private static function maskMobileNumber($mobileNumber) {
        if (strlen($mobileNumber) >= 4) {
            return substr($mobileNumber, 0, 2) . str_repeat('*', strlen($mobileNumber) - 4) . substr($mobileNumber, -2);
        }
        return str_repeat('*', strlen($mobileNumber));
    }
}
```

### Error Handling for Digital Wallets

```php
/**
 * Digital wallet specific error handling
 */
class DigitalWalletErrorHandler {
    
    private static $walletErrorCodes = [
        'URPAY_INVALID_MOBILE' => [
            'message' => 'Invalid URPAY mobile number',
            'solution' => 'Please enter a valid 9-digit mobile number registered with URPAY'
        ],
        'URPAY_INSUFFICIENT_BALANCE' => [
            'message' => 'Insufficient URPAY wallet balance',
            'solution' => 'Please top up your URPAY wallet or use a different payment method'
        ],
        'URPAY_TRANSACTION_LIMIT' => [
            'message' => 'Transaction exceeds URPAY limit',
            'solution' => 'Please contact URPAY support to increase your transaction limit'
        ],
        'APPLE_PAY_NOT_SUPPORTED' => [
            'message' => 'Apple Pay is not supported on this device',
            'solution' => 'Please use a different payment method'
        ],
        'APPLE_PAY_SETUP_REQUIRED' => [
            'message' => 'Apple Pay is not set up',
            'solution' => 'Please set up Apple Pay in your device settings'
        ]
    ];
    
    public static function handleWalletError($errorCode, $walletType, $context = []) {
        $errorInfo = self::$walletErrorCodes[$errorCode] ?? [
            'message' => 'Digital wallet payment failed',
            'solution' => 'Please try again or use a different payment method'
        ];
        
        // Log wallet-specific error
        error_log("Wallet Error [{$walletType}]: {$errorCode} - " . json_encode($context));
        
        return [
            'success' => false,
            'error_code' => $errorCode,
            'wallet_type' => $walletType,
            'user_message' => $errorInfo['message'],
            'suggested_solution' => $errorInfo['solution'],
            'timestamp' => date('Y-m-d H:i:s')
        ];
    }
}
```
