# PHP Encryption & Code Examples

This section provides comprehensive PHP code examples for integrating with Al Rajhi Bank Payment Gateway, including encryption, decryption, and complete API implementations.

## Complete PHP SDK

### ARB Payment Gateway Class

```php
<?php

/**
 * Al Rajhi Bank Payment Gateway PHP SDK
 * 
 * @version 1.24
 * @author Al Rajhi Bank Integration Team
 */
class ARBPaymentGateway {
    
    private $tranportalId;
    private $tranportalPassword;
    private $resourceKey;
    private $environment;
    private $baseUrl;
    
    // Environment URLs
    const SANDBOX_URL = 'https://sandbox.digitalpayments.alrajhibank.com.sa';
    const PRODUCTION_URL = 'https://digitalpayments.alrajhibank.com.sa';
    
    // API Endpoints
    const PAYMENT_TOKEN_ENDPOINT = '/pg/payment/tranportal/api/paymentToken.htm';
    const TRANSACTION_STATUS_ENDPOINT = '/pg/payment/tranportal/api/transactionStatus.htm';
    const REFUND_ENDPOINT = '/pg/payment/tranportal/api/refund.htm';
    const INQUIRY_ENDPOINT = '/pg/payment/tranportal/api/inquiry.htm';
    
    /**
     * Constructor
     * 
     * @param array $config Configuration array
     */
    public function __construct($config) {
        $this->tranportalId = $config['tranportal_id'];
        $this->tranportalPassword = $config['tranportal_password'];
        $this->resourceKey = $config['resource_key'];
        $this->environment = $config['environment'] ?? 'sandbox';
        
        $this->baseUrl = $this->environment === 'production' 
            ? self::PRODUCTION_URL 
            : self::SANDBOX_URL;
    }
    
    /**
     * Encrypt data using AES-256-CBC
     * 
     * @param array $data Data to encrypt
     * @return string Base64 encoded encrypted data
     */
    public function encrypt($data) {
        // Convert data to JSON
        $jsonData = json_encode($data);
        
        // URL encode the JSON data before encryption
        $encodedData = urlencode($jsonData);
        
        // Encryption parameters
        $method = 'AES-256-CBC';
        $key = $this->resourceKey;
        $iv = substr($key, 0, 16); // Use first 16 characters as IV
        
        // Encrypt the data
        $encrypted = openssl_encrypt(
            $encodedData,
            $method,
            $key,
            OPENSSL_RAW_DATA,
            $iv
        );
        
        // Return base64 encoded encrypted data
        return base64_encode($encrypted);
    }
    
    /**
     * Decrypt data using AES-256-CBC
     * 
     * @param string $encryptedData Base64 encoded encrypted data
     * @return array Decrypted data as array
     */
    public function decrypt($encryptedData) {
        // Decode base64
        $data = base64_decode($encryptedData);
        
        // Decryption parameters
        $method = 'AES-256-CBC';
        $key = $this->resourceKey;
        $iv = substr($key, 0, 16); // Use first 16 characters as IV
        
        // Decrypt the data
        $decrypted = openssl_decrypt(
            $data,
            $method,
            $key,
            OPENSSL_RAW_DATA,
            $iv
        );
        
        // URL decode the decrypted data
        $decodedData = urldecode($decrypted);
        
        // Return as array
        return json_decode($decodedData, true);
    }
    
    /**
     * Make HTTP request to ARB Payment Gateway
     * 
     * @param string $endpoint API endpoint
     * @param array $data Request data
     * @return array Response data
     */
    private function makeRequest($endpoint, $data) {
        $url = $this->baseUrl . $endpoint;
        
        // Initialize cURL
        $ch = curl_init();
        
        // Set cURL options
        curl_setopt_array($ch, [
            CURLOPT_URL => $url,
            CURLOPT_POST => true,
            CURLOPT_POSTFIELDS => json_encode($data),
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_SSL_VERIFYPEER => true,
            CURLOPT_SSL_VERIFYHOST => 2,
            CURLOPT_TIMEOUT => 30,
            CURLOPT_HTTPHEADER => [
                'Content-Type: application/json',
                'Accept: application/json'
            ]
        ]);
        
        // Execute request
        $response = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        $error = curl_error($ch);
        
        curl_close($ch);
        
        // Handle cURL errors
        if ($error) {
            throw new Exception("cURL Error: " . $error);
        }
        
        // Handle HTTP errors
        if ($httpCode !== 200) {
            throw new Exception("HTTP Error: " . $httpCode);
        }
        
        // Decode response
        $responseData = json_decode($response, true);
        
        if (json_last_error() !== JSON_ERROR_NONE) {
            throw new Exception("Invalid JSON response: " . json_last_error_msg());
        }
        
        return $responseData;
    }
    
    /**
     * Create payment token for Bank Hosted integration
     * 
     * @param array $params Payment parameters
     * @return array Payment response with payment ID and URL
     */
    public function createPaymentToken($params) {
        // Prepare transaction data
        $trandata = [
            'amt' => $params['amt'],
            'action' => $params['action'] ?? '1', // 1 = Purchase
            'password' => $this->tranportalPassword,
            'id' => $this->tranportalId,
            'currencyCode' => $params['currencyCode'] ?? '682', // SAR
            'trackId' => $params['trackId'] ?? uniqid('TXN_'),
            'responseURL' => $params['responseURL'],
            'errorURL' => $params['errorURL'] ?? $params['responseURL']
        ];
        
        // Add optional parameters
        $optionalParams = ['udf1', 'udf2', 'udf3', 'udf4', 'udf5', 'langid'];
        foreach ($optionalParams as $param) {
            if (isset($params[$param])) {
                $trandata[$param] = $params[$param];
            }
        }
        
        // Encrypt transaction data
        $encryptedTrandata = $this->encrypt($trandata);
        
        // Prepare request
        $requestData = [
            'id' => $this->tranportalId,
            'trandata' => $encryptedTrandata,
            'responseURL' => $params['responseURL'],
            'errorURL' => $params['errorURL'] ?? $params['responseURL']
        ];
        
        // Make API request
        $response = $this->makeRequest(self::PAYMENT_TOKEN_ENDPOINT, $requestData);
        
        // Process response
        if (isset($response['status']) && $response['status'] == '1') {
            return [
                'success' => true,
                'paymentId' => $response['paymentId'],
                'paymentURL' => $response['paymentURL'],
                'trackId' => $trandata['trackId']
            ];
        } else {
            return [
                'success' => false,
                'error' => $response['errorText'] ?? 'Unknown error occurred',
                'errorCode' => $response['error'] ?? 'UNKNOWN'
            ];
        }
    }
    
    /**
     * Check transaction status
     * 
     * @param string $paymentId Payment ID from initial response
     * @return array Transaction status
     */
    public function getTransactionStatus($paymentId) {
        // Prepare transaction data
        $trandata = [
            'password' => $this->tranportalPassword,
            'id' => $this->tranportalId,
            'paymentId' => $paymentId,
            'action' => '10' // Inquiry action
        ];
        
        // Encrypt transaction data
        $encryptedTrandata = $this->encrypt($trandata);
        
        // Prepare request
        $requestData = [
            'id' => $this->tranportalId,
            'trandata' => $encryptedTrandata
        ];
        
        // Make API request
        $response = $this->makeRequest(self::TRANSACTION_STATUS_ENDPOINT, $requestData);
        
        // Decrypt response if trandata is present
        if (isset($response['trandata']) && !empty($response['trandata'])) {
            $decryptedData = $this->decrypt($response['trandata']);
            return [
                'success' => true,
                'data' => $decryptedData
            ];
        }
        
        return [
            'success' => false,
            'error' => $response['errorText'] ?? 'Unable to get transaction status',
            'errorCode' => $response['error'] ?? 'UNKNOWN'
        ];
    }
    
    /**
     * Process refund
     * 
     * @param array $params Refund parameters
     * @return array Refund response
     */
    public function processRefund($params) {
        // Prepare transaction data
        $trandata = [
            'password' => $this->tranportalPassword,
            'id' => $this->tranportalId,
            'action' => '2', // Refund action
            'amt' => $params['amt'],
            'currencyCode' => $params['currencyCode'] ?? '682',
            'trackId' => $params['trackId'],
            'transId' => $params['transId'] // Original transaction ID
        ];
        
        // Encrypt transaction data
        $encryptedTrandata = $this->encrypt($trandata);
        
        // Prepare request
        $requestData = [
            'id' => $this->tranportalId,
            'trandata' => $encryptedTrandata
        ];
        
        // Make API request
        $response = $this->makeRequest(self::REFUND_ENDPOINT, $requestData);
        
        // Process response
        if (isset($response['trandata']) && !empty($response['trandata'])) {
            $decryptedData = $this->decrypt($response['trandata']);
            return [
                'success' => true,
                'data' => $decryptedData
            ];
        }
        
        return [
            'success' => false,
            'error' => $response['errorText'] ?? 'Refund failed',
            'errorCode' => $response['error'] ?? 'UNKNOWN'
        ];
    }
    
    /**
     * Handle callback response from payment gateway
     * 
     * @param array $callbackData POST data from callback
     * @return array Processed callback data
     */
    public function handleCallback($callbackData) {
        try {
            // Check if trandata exists and is not empty
            if (isset($callbackData['trandata']) && !empty($callbackData['trandata'])) {
                // Decrypt transaction data
                $decryptedData = $this->decrypt($callbackData['trandata']);
                
                // Determine transaction status
                $isSuccess = false;
                if (isset($decryptedData['result'])) {
                    $isSuccess = in_array(strtoupper($decryptedData['result']), [
                        'CAPTURED', 'APPROVED'
                    ]);
                }
                
                return [
                    'success' => $isSuccess,
                    'data' => $decryptedData,
                    'paymentId' => $callbackData['paymentid'] ?? null,
                    'transactionId' => $callbackData['tranid'] ?? null
                ];
            } else {
                // Handle error response
                return [
                    'success' => false,
                    'error' => $callbackData['ErrorText'] ?? 'Transaction failed',
                    'errorCode' => $callbackData['Error'] ?? 'UNKNOWN',
                    'paymentId' => $callbackData['paymentid'] ?? null,
                    'transactionId' => $callbackData['tranid'] ?? null
                ];
            }
        } catch (Exception $e) {
            return [
                'success' => false,
                'error' => 'Failed to process callback: ' . $e->getMessage(),
                'errorCode' => 'CALLBACK_ERROR'
            ];
        }
    }
}
```

## Usage Examples

### Basic Payment Processing

```php
<?php
// Initialize the gateway
$gateway = new ARBPaymentGateway([
    'tranportal_id' => 'YOUR_TRANPORTAL_ID',
    'tranportal_password' => 'YOUR_TRANPORTAL_PASSWORD',
    'resource_key' => 'YOUR_RESOURCE_KEY',
    'environment' => 'sandbox' // or 'production'
]);

// Create payment token
$paymentParams = [
    'amt' => '100.00',
    'currencyCode' => '682', // SAR
    'trackId' => 'ORDER_' . time(),
    'responseURL' => 'https://yoursite.com/payment-callback.php',
    'errorURL' => 'https://yoursite.com/payment-error.php',
    'udf1' => 'Customer ID: 12345',
    'udf2' => 'Order ID: ORD001'
];

try {
    $response = $gateway->createPaymentToken($paymentParams);
    
    if ($response['success']) {
        // Redirect customer to payment page
        $paymentURL = $response['paymentURL'] . '?paymentid=' . $response['paymentId'];
        header('Location: ' . $paymentURL);
        exit;
    } else {
        echo "Payment initialization failed: " . $response['error'];
    }
} catch (Exception $e) {
    echo "Error: " . $e->getMessage();
}
```

### Handling Payment Callback

```php
<?php
// payment-callback.php
require_once 'ARBPaymentGateway.php';

$gateway = new ARBPaymentGateway([
    'tranportal_id' => 'YOUR_TRANPORTAL_ID',
    'tranportal_password' => 'YOUR_TRANPORTAL_PASSWORD',
    'resource_key' => 'YOUR_RESOURCE_KEY',
    'environment' => 'sandbox'
]);

// Process callback
$callbackResult = $gateway->handleCallback($_POST);

if ($callbackResult['success']) {
    // Payment successful
    $transactionData = $callbackResult['data'];
    
    echo "<h1>Payment Successful!</h1>";
    echo "<p>Transaction ID: " . $transactionData['tranid'] . "</p>";
    echo "<p>Amount: " . $transactionData['amt'] . " " . $transactionData['currencyCode'] . "</p>";
    echo "<p>Track ID: " . $transactionData['trackId'] . "</p>";
    
    // Update your database, send confirmation email, etc.
    
} else {
    // Payment failed
    echo "<h1>Payment Failed</h1>";
    echo "<p>Error: " . $callbackResult['error'] . "</p>";
    echo "<p>Error Code: " . $callbackResult['errorCode'] . "</p>";
}
```

## Advanced Features

### Custom Encryption Helper

```php
<?php
/**
 * Standalone encryption helper for ARB Payment Gateway
 */
class ARBEncryption {
    
    private $resourceKey;
    
    public function __construct($resourceKey) {
        $this->resourceKey = $resourceKey;
    }
    
    /**
     * Encrypt data with URL encoding
     */
    public function encryptData($data) {
        $jsonData = json_encode($data);
        $encodedData = urlencode($jsonData);
        
        $encrypted = openssl_encrypt(
            $encodedData,
            'AES-256-CBC',
            $this->resourceKey,
            OPENSSL_RAW_DATA,
            substr($this->resourceKey, 0, 16)
        );
        
        return base64_encode($encrypted);
    }
    
    /**
     * Decrypt data with URL decoding
     */
    public function decryptData($encryptedData) {
        $data = base64_decode($encryptedData);
        
        $decrypted = openssl_decrypt(
            $data,
            'AES-256-CBC',
            $this->resourceKey,
            OPENSSL_RAW_DATA,
            substr($this->resourceKey, 0, 16)
        );
        
        $decodedData = urldecode($decrypted);
        return json_decode($decodedData, true);
    }
}

// Usage
$encryption = new ARBEncryption('YOUR_RESOURCE_KEY');

$data = ['amt' => '100.00', 'action' => '1'];
$encrypted = $encryption->encryptData($data);
$decrypted = $encryption->decryptData($encrypted);
```

### Error Handling Utility

```php
<?php
/**
 * ARB Payment Gateway Error Handler
 */
class ARBErrorHandler {
    
    // Common error codes and descriptions
    private static $errorCodes = [
        'IPAY0100114' => 'Duplicate Record, transaction ID already exists',
        'IPAY0100119' => 'Transaction denied due to invalid card number',
        'IPAY0100125' => 'Payment instrument not enabled',
        'IPAY0100061' => 'Transaction denied due to invalid amount',
        'IPAY0100042' => 'Transaction time limit exceeds',
        // Add more error codes as needed
    ];
    
    /**
     * Get error description by code
     */
    public static function getErrorDescription($errorCode) {
        return self::$errorCodes[$errorCode] ?? 'Unknown error occurred';
    }
    
    /**
     * Log error for debugging
     */
    public static function logError($errorCode, $errorText, $context = []) {
        $logEntry = [
            'timestamp' => date('Y-m-d H:i:s'),
            'error_code' => $errorCode,
            'error_text' => $errorText,
            'context' => $context
        ];
        
        // Log to file or database
        error_log(json_encode($logEntry), 3, 'arb_payment_errors.log');
    }
    
    /**
     * Handle and format error response
     */
    public static function handleError($errorCode, $errorText = null) {
        $description = self::getErrorDescription($errorCode);
        
        self::logError($errorCode, $errorText, debug_backtrace());
        
        return [
            'success' => false,
            'error_code' => $errorCode,
            'error_text' => $errorText ?: $description,
            'error_description' => $description
        ];
    }
}
```

::callout{icon="i-lucide-info" color="blue"}
**Important Notes:**
- Always use URL encoding before encryption and URL decoding after decryption
- Store your Resource Key securely and never expose it in client-side code
- Use HTTPS for all payment-related communications
- Implement proper error handling and logging for production environments
