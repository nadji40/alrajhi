# Merchant Hosted Integration

Merchant Hosted Integration allows you to process payments directly on your website with complete control over the user experience. This method provides maximum customization capabilities while maintaining security through proper implementation.

## Overview

In this integration method, customers enter their payment details on your website, and the payment data is processed directly through Al Rajhi Bank Payment Gateway APIs. This approach offers the most flexibility but requires careful attention to security and PCI compliance.

### Key Benefits

- **Complete UI Control**: Design custom payment forms that match your brand
- **Enhanced User Experience**: Seamless checkout without redirects
- **Advanced Features**: Support for card-on-file, recurring payments, and tokenization
- **Real-time Processing**: Immediate payment results and status updates
- **Mobile App Integration**: Perfect for mobile applications and SPAs

### Security Considerations

- **PCI Compliance**: Your application must meet PCI-DSS requirements
- **Data Encryption**: All sensitive data must be encrypted before transmission
- **Secure Storage**: Never store card details in plain text
- **HTTPS Required**: All payment pages must use SSL/TLS encryption

### Step 1: Secure Payment Form

```php
<?php
/**
 * Merchant Hosted Payment Form Handler
 */
class MerchantHostedPayment {
    private $gateway;
    private $validator;
    
    public function __construct($gateway, $validator) {
        $this->gateway = $gateway;
        $this->validator = $validator;
    }
    
    /**
     * Process direct payment
     */
    public function processDirectPayment($paymentData) {
        try {
            // Step 1: Validate input data
            $validation = $this->validator->validatePaymentData($paymentData);
            if (!$validation['valid']) {
                throw new Exception('Validation failed: ' . implode(', ', $validation['errors']));
            }
            
            // Step 2: Prepare transaction parameters
            $transactionParams = [
                'amt' => $paymentData['amount'],
                'action' => '1', // Purchase
                'password' => $this->gateway->getCredentials()['password'],
                'id' => $this->gateway->getCredentials()['tranportal_id'],
                'currencyCode' => $paymentData['currency'] ?? '682',
                'trackId' => $paymentData['order_id'],
                'card_number' => $paymentData['card_number'],
                'expiry_date' => $paymentData['expiry_date'],
                'cvv' => $paymentData['cvv'],
                'cardholder_name' => $paymentData['cardholder_name'],
                'responseURL' => $paymentData['callback_url']
            ];
            
            // Step 3: Add optional parameters
            if (isset($paymentData['billing_address'])) {
                $transactionParams['address'] = $paymentData['billing_address'];
                $transactionParams['postal_code'] = $paymentData['postal_code'];
            }
            
            // Step 4: Process payment
            $response = $this->gateway->processDirectPayment($transactionParams);
            
            // Step 5: Handle response
            return $this->handlePaymentResponse($response, $paymentData);
            
        } catch (Exception $e) {
            return [
                'success' => false,
                'error' => $e->getMessage(),
                'error_code' => 'PROCESSING_ERROR'
            ];
        }
    }
    
    private function handlePaymentResponse($response, $originalData) {
        if ($response['success']) {
            // Payment successful
            $transactionData = $response['data'];
            
            // Update order status
            $this->updateOrderStatus($originalData['order_id'], 'completed', $transactionData);
            
            // Send confirmation
            $this->sendPaymentConfirmation($originalData['customer_email'], $transactionData);
            
            return [
                'success' => true,
                'transaction_id' => $transactionData['tranid'],
                'amount' => $transactionData['amt'],
                'status' => $transactionData['result'],
                'reference' => $transactionData['trackId']
            ];
        } else {
            // Payment failed
            $this->updateOrderStatus($originalData['order_id'], 'failed', $response);
            
            return [
                'success' => false,
                'error' => $response['error'],
                'error_code' => $response['error_code']
            ];
        }
    }

        // Implementation depends on your database structure
    }
    
    private function sendPaymentConfirmation($email, $transactionData) {

        // Implementation depends on your email system
    }
}
```

```html
                    errorElement.textContent = 'Card number must be 13-19 digits';
                    return false;
                }
                
                // Luhn algorithm validation
                if (!isValidCardNumber(cardNumber)) {
                    errorElement.textContent = 'Invalid card number';
                    return false;

                errorElement.textContent = '';
                return true;
            }
            
            function validateExpiryDate(expiryDate) {
                const errorElement = document.getElementById('expiryError');
                
                if (!/^\d{2}\/\d{4}$/.test(expiryDate)) {
                    errorElement.textContent = 'Invalid format (MM/YYYY)';
                    return false;
                }
                
                const [month, year] = expiryDate.split('/');
                const currentDate = new Date();
                const currentYear = currentDate.getFullYear();
                const currentMonth = currentDate.getMonth() + 1;
                
                if (parseInt(month) < 1 || parseInt(month) > 12) {
                    errorElement.textContent = 'Invalid month';
                    return false;
                }
                
                if (parseInt(year) < currentYear || 
                    (parseInt(year) === currentYear && parseInt(month) < currentMonth)) {
                    errorElement.textContent = 'Card has expired';
                    return false;
                }
                
                errorElement.textContent = '';
                return true;
            }
            
            function validateCVV(cvv) {
                const errorElement = document.getElementById('cvvError');
                
                if (cvv.length < 3 || cvv.length > 4) {
                    errorElement.textContent = 'CVV must be 3-4 digits';
                    return false;
                }
                
                errorElement.textContent = '';
                return true;
            }
            
            function validateForm() {
                const cardNumber = cardNumberInput.value.replace(/\s/g, '');
                const expiryDate = expiryInput.value;
                const cvv = cvvInput.value;
                
                return validateCardNumber(cardNumber) && 
                       validateExpiryDate(expiryDate) && 
                       validateCVV(cvv);
            }
            
            function isValidCardNumber(cardNumber) {
                let sum = 0;
                let alternate = false;
                
                for (let i = cardNumber.length - 1; i >= 0; i--) {
                    let digit = parseInt(cardNumber.charAt(i));
                    
                    if (alternate) {
                        digit *= 2;
                        if (digit > 9) {
                            digit = (digit % 10) + 1;
                        }
                    }
                    
                    sum += digit;
                    alternate = !alternate;
                }
                
                return (sum % 10) === 0;
            }
        });
```

#### Card-on-File Implementation

```php
/**
 * Card-on-File functionality for repeat customers
 */
class CardOnFileManager {
    private $gateway;
    private $database;
    
    public function __construct($gateway, $database) {
        $this->gateway = $gateway;
        $this->database = $database;
    }
    
    /**
     * Register card for future use
     */
    public function registerCard($customerId, $cardData) {
        $registrationParams = [
            'action' => '11', // Card registration
            'password' => $this->gateway->getCredentials()['password'],
            'id' => $this->gateway->getCredentials()['tranportal_id'],
            'trackId' => 'CARD_REG_' . time(),
            'card_number' => $cardData['card_number'],
            'expiry_date' => $cardData['expiry_date'],
            'cardholder_name' => $cardData['cardholder_name'],
            'customer_id' => $customerId
        ];
        
        $response = $this->gateway->registerCard($registrationParams);
        
        if ($response['success']) {
            // Store card token securely
            $this->storeCardToken($customerId, $response['card_token'], $cardData);
            
            return [
                'success' => true,
                'card_token' => $response['card_token'],
                'masked_card' => $this->maskCardNumber($cardData['card_number'])
            ];
        }
        
        return $response;
    }
    
    /**
     * Process payment using stored card
     */
    public function processStoredCardPayment($customerId, $cardToken, $amount, $orderId) {
        $paymentParams = [
            'amt' => $amount,
            'action' => '1', // Purchase
            'password' => $this->gateway->getCredentials()['password'],
            'id' => $this->gateway->getCredentials()['tranportal_id'],
            'currencyCode' => '682',
            'trackId' => $orderId,
            'card_token' => $cardToken,
            'customer_id' => $customerId
        ];
        
        return $this->gateway->processTokenPayment($paymentParams);
    }
    
    /**
     * Get customer's stored cards
     */
    public function getCustomerCards($customerId) {
        return $this->database->getStoredCards($customerId);
    }
    
    /**
     * Remove stored card
     */
    public function removeStoredCard($customerId, $cardToken) {
        $deregistrationParams = [
            'action' => '12', // Card deregistration
            'password' => $this->gateway->getCredentials()['password'],
            'id' => $this->gateway->getCredentials()['tranportal_id'],
            'card_token' => $cardToken,
            'customer_id' => $customerId
        ];
        
        $response = $this->gateway->deregisterCard($deregistrationParams);
        
        if ($response['success']) {
            $this->database->removeStoredCard($customerId, $cardToken);
        }
        
        return $response;
    }
    
    private function storeCardToken($customerId, $cardToken, $cardData) {
        $cardInfo = [
            'customer_id' => $customerId,
            'card_token' => $cardToken,
            'masked_card' => $this->maskCardNumber($cardData['card_number']),
            'card_type' => $this->detectCardType($cardData['card_number']),
            'expiry_date' => $cardData['expiry_date'],
            'created_at' => date('Y-m-d H:i:s')
        ];
        
        $this->database->storeCardToken($cardInfo);
    }
    
    private function maskCardNumber($cardNumber) {
        $cleaned = preg_replace('/\s/', '', $cardNumber);
        return substr($cleaned, 0, 4) . str_repeat('*', strlen($cleaned) - 8) . substr($cleaned, -4);
    }
    
    private function detectCardType($cardNumber) {
        $cleaned = preg_replace('/\s/', '', $cardNumber);
        
        if (preg_match('/^4/', $cleaned)) {
            return 'Visa';
        } elseif (preg_match('/^5[1-5]/', $cleaned)) {
            return 'MasterCard';
        } elseif (preg_match('/^3[47]/', $cleaned)) {
            return 'American Express';
        }
        
        return 'Unknown';
    }
}
```

### Implementation Security

```php
/**
 * Security utilities for merchant hosted integration
 */
class PaymentSecurity {
    
    /**
     * Sanitize and validate input data
     */
    public static function sanitizePaymentData($data) {
        $sanitized = [];
        
        // Sanitize card number
        if (isset($data['card_number'])) {
            $sanitized['card_number'] = preg_replace('/[^0-9]/', '', $data['card_number']);
        }
        
        // Sanitize expiry date
        if (isset($data['expiry_date'])) {
            $sanitized['expiry_date'] = preg_replace('/[^0-9\/]/', '', $data['expiry_date']);
        }
        
        // Sanitize CVV
        if (isset($data['cvv'])) {
            $sanitized['cvv'] = preg_replace('/[^0-9]/', '', $data['cvv']);
        }
        
        // Sanitize cardholder name
        if (isset($data['cardholder_name'])) {
            $sanitized['cardholder_name'] = filter_var($data['cardholder_name'], FILTER_SANITIZE_STRING);
        }
        
        return $sanitized;
    }
    
    /**
     * Generate secure transaction ID
     */
    public static function generateSecureTransactionId() {
        return 'TXN_' . bin2hex(random_bytes(16)) . '_' . time();
    }
    
    /**
     * Log security events
     */
    public static function logSecurityEvent($event, $data = []) {
        $logEntry = [
            'timestamp' => date('Y-m-d H:i:s'),
            'event' => $event,
            'ip_address' => $_SERVER['REMOTE_ADDR'] ?? 'Unknown',
            'user_agent' => $_SERVER['HTTP_USER_AGENT'] ?? 'Unknown',
            'data' => self::sanitizeLogData($data)
        ];
        
        error_log('Security Event: ' . json_encode($logEntry), 3, 'security_events.log');
    }
    
    private static function sanitizeLogData($data) {
        $sensitiveFields = ['card_number', 'cvv', 'password'];
        
        array_walk_recursive($data, function(&$value, $key) use ($sensitiveFields) {
            if (in_array(strtolower($key), $sensitiveFields)) {
                $value = '***REDACTED***';
            }
        });
        
        return $data;
    }
}
```
